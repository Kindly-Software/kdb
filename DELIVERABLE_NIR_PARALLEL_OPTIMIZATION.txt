USAGE EXAMPLE
================================================================================

use atomic_capsule::gpu::{NIRParallelOptimizationCapsule, ShaderStage};

// Create capsule
let capsule = NIRParallelOptimizationCapsule::new();

// Submit 3-stage pipeline
capsule.submit_stage(ShaderStage::Vertex, 100)?;     // 100 instructions
capsule.submit_stage(ShaderStage::Fragment, 150)?;   // 150 instructions
capsule.submit_stage(ShaderStage::Compute, 200)?;    // 200 instructions

// Optimize in parallel
capsule.optimize_parallel()?;

// Get results
let vs_result = capsule.get_result(ShaderStage::Vertex)?;
assert_eq!(vs_result.instructions_after, 80);   // 20% reduction
assert_eq!(vs_result.reduction_percent, 20);

// Reset for next cycle
capsule.reset();
