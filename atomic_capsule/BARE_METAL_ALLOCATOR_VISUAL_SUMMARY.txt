Error Types:
  • AllocError::InvalidSize (size is zero or invalid)
  • AllocError::SizeTooLarge (exceeds maximum size class)
  • AllocError::OutOfMemory (no free blocks available)
  • AllocError::AlignmentFailed (cannot satisfy alignment)
  • AllocError::NoContiguousSpace (cannot find contiguous space)

🚀 USAGE EXAMPLE
────────────────────────────────────────────────────────────────────────────
use atomic_capsule::gpu::kgpu_driver::BareMetalAllocatorCapsule;

// Initialize pools
let pools = [
    MemoryPool::new(0x1000_0000, 256*1024*1024, 4096, PoolType::Vram),
    MemoryPool::new(0x2000_0000, 64*1024*1024, 4096, PoolType::SystemVisible),
    MemoryPool::new(0, 0, 0, PoolType::Vram),
    MemoryPool::new(0, 0, 0, PoolType::Vram),
];

let allocator = BareMetalAllocatorCapsule::new(pools);
unsafe { allocator.initialize(); }

// Allocate command buffer (4K, 4K-aligned)
let cmd_buf = allocator.alloc(4096, 4096)?;

// Allocate framebuffer (1080p RGBA, contiguous)
let fb = allocator.alloc_contiguous(1920 * 1080 * 4)?;

// Get statistics
let stats = allocator.get_stats();
println!("Allocated: {} bytes", stats.bytes_in_use);

// Free when done
unsafe {
    allocator.free(cmd_buf, 4096);
    allocator.free(fb, 1920 * 1080 * 4);
}
