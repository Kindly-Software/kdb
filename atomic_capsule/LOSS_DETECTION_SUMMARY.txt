================================================================================
INTEGRATION STEPS FOR END USERS
================================================================================

1. Enable feature flag in Cargo.toml:
   atomic_capsule = { version = "0.8.0", features = ["quic"] }

2. Import the capsule:
   use atomic_capsule::quic::LossDetectionCapsule;

3. Create per-QUIC-connection instance:
   let loss_detector = LossDetectionCapsule::new();

4. Update on each ACK:
   loss_detector.update_rtt(ack_rtt_ns);

5. Check for packet loss:
   if loss_detector.is_packet_lost(packet_age_ns, ack_count) {
       // Retransmit
   }

6. Compute retransmission timeout:
   let rto_ns = loss_detector.compute_rto_ns();

Memory cost per connection: 128 bytes (negligible)
