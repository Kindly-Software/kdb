================================================================================
USAGE EXAMPLE
================================================================================

use atomic_capsule::tls::{TlsAlpnCapsule, Protocol};

// Create capsule
let alpn = TlsAlpnCapsule::new(&["h2", "http/1.1"])?;

// Handle TLS handshake
let client_alpn = vec![b"h2".to_vec(), b"http/1.1".to_vec()];
match alpn.negotiate(&client_alpn) {
    Ok(Protocol::Http2) => handle_http2(),      // Server pref: h2
    Ok(Protocol::Http11) => handle_http11(),    // Fallback
    Err(_) => reject_connection(),              // No match
}

// Monitor metrics
let metrics = alpn.get_metrics();
println!("H2: {}, HTTP/1.1: {}, Success: {:.1}%",
    metrics.h2_count, metrics.http11_count, metrics.success_rate);
