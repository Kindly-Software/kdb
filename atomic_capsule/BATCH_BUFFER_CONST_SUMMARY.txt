================================================================================
  USAGE EXAMPLE
================================================================================

use atomic_capsule::parallel::BatchBufferConst;

// Create buffer (zero allocation, compile-time batch size)
let buffer = BatchBufferConst::<i32, 64>::new();

// Thread-local push (no synchronization)
for i in 0..100 {
    match buffer.push(i) {
        Ok(Some(batch)) => {
            // Batch full (64 items) - process
            println!("Processing {} items", batch.len());
            for item in batch.iter() {
                println!("{}", item);
            }
        },
        Ok(None) => {
            // Item buffered, continue
        },
        Err(e) => {
            eprintln!("Error: {}", e);
        }
    }
}

// Manual flush for remaining items
if let Ok(batch) = buffer.flush() {
    println!("Final batch: {} items", batch.len());
}
