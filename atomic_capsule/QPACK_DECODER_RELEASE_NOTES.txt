USAGE PATTERNS
================================================================================

Basic Decoding:
  let decoder = QpackDecoderCapsule::new(4096);
  let headers = decoder.decode_headers(&encoded)?;
  
  for (name, value) in headers {
      println!("{}: {}", name, value);
  }

Batch Processing:
  let packets = vec![
      vec![0x82],          // :method GET
      vec![0x85],          // :scheme https
      vec![0x82, 0x85],    // Both
  ];
  
  let results = decoder.decode_batch(&packets);
  for (idx, result) in results.iter().enumerate() {
      match result {
          Ok(headers) => println!("Packet {}: {} headers", idx, headers.len()),
          Err(e) => eprintln!("Packet {} error: {}", idx, e),
      }
  }

Metrics:
  let (headers, bytes, known) = decoder.metrics();
  println!("Total: {} headers, {} bytes, known count: {}", headers, bytes, known);

Dynamic Table Sync:
  decoder.set_known_received_count(10); // Encoder ACK update
  let (_, _, known) = decoder.metrics();
  assert_eq!(known, 10);
