USAGE PATTERNS
================================================================================

Pattern 1: Web Server (select protocol per connection)
  for conn in listener.accept() {
      match alpn.negotiate(&conn.client_alpn()) {
          Ok(proto) => handle_protocol(conn, proto),
          Err(_) => reject(conn),
      }
  }

Pattern 2: Load Balancer (metrics collection)
  let metrics = alpn.get_metrics();
  println!("H2: {:.1}%, HTTP/1.1: {:.1}%",
      h2_pct, http11_pct);

  if feature_http2_deprecated {

  }

AlpnError::EmptyProtocolList
  → Zero protocols provided

All errors implement Display + Error traits
