USAGE EXAMPLE
═════════════════════════════════════════════════════════════════════════════

use atomic_capsule::tui::FileNavigatorCapsule;
use std::path::PathBuf;

fn main() -> std::io::Result<()> {
    // Create navigator
    let mut nav = FileNavigatorCapsule::new(PathBuf::from("/home"));
    
    // Refresh directory
    nav.refresh(&PathBuf::from("/home"))?;
    println!("Directory has {} entries", nav.total_entries());
    
    // Navigate
    nav.navigate_down();  // Move to next entry (<10ns)
    println!("Selected: {}", nav.current_index());
    
    // Detect changes
    if nav.current_dir_changed() {
        nav.refresh(&PathBuf::from("/home"))?;
    }
    
    // Filter files
    nav.set_filter_flags(filter_flags::HIDE_HIDDEN);  // Hide . files
    
    Ok(())
}
