DELETION PROOF GENERATION:
──────────────────────────
    let mut capsule = DeletionProofCapsule::new(user_id, session_id)?;
    // ... record snapshots ...

    // Generate cryptographic proof of deletion (two-phase commit)
    let cert = capsule.generate_deletion_proof(
        &server_private_key,
        &Path::new("/user/data/dir")
    )?;

    // Save certificate for user
    std::fs::write("deletion_proof.json", cert.to_json()?)?;

CLIENT-SIDE VERIFICATION:
──────────────────────────
    // User can verify offline (no server required!)
    let json = std::fs::read_to_string("deletion_proof.json")?;
    let cert = DeletionCertificate::from_json(&json)?;

    DeletionProofCapsule::verify_certificate(&cert, &server_public_key)?;
    println!("✓ {} bytes permanently deleted", cert.bytes_deleted);
