USAGE EXAMPLE
================================================================================

use kindly_verified_web::capsules::{
    ZeroTrustSessionCapsule, SessionState, RequestMetadata, calculate_risk_score,
};

// Create session
let capsule = ZeroTrustSessionCapsule::new(
    token_hash,          // Session token hash
    user_id,             // From JWT/OAuth2
    device_fp,           // User-Agent, canvas fp
    ip_hash,             // Privacy-preserving IP
    current_ts,          // Microseconds since epoch
);

// Risk scoring (5 behavioral signals)
let metadata = RequestMetadata {
    ip_changed: false,
    device_changed: false,
    unusual_time: false,
    unusual_location: false,
    failed_verification_rate: 0.0,
};

let risk_score = calculate_risk_score(&metadata);
capsule.update_risk_score(risk_score, current_ts);

// Check if verification needed
if capsule.needs_verification(current_ts) {
    match perform_verification(&capsule) {
        Ok(_) => capsule.record_verification_success(),
        Err(_) => {
            capsule.record_verification_failure();
            capsule.transition_state(
                SessionState::Active,
                SessionState::Challenged,
                current_ts
            );
        }
    }
}

// Get state
match capsule.get_state() {
    SessionState::Active => { /* Allow access */ }
    SessionState::Challenged => { /* Require MFA */ }
    SessionState::Suspended => { /* Block access */ }
    SessionState::Expired => { /* Redirect to login */ }
}
